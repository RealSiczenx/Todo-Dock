cmake_minimum_required(VERSION 3.24)
project(task-dock VERSION 1.0.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

option(TASK_DOCK_TESTS_ONLY "Build portable logic tests without OBS or Qt" OFF)
if(TASK_DOCK_TESTS_ONLY)
  enable_testing()
  add_executable(task-dock-tests tests/logic-test.cpp)
  add_test(NAME task-dock-logic COMMAND task-dock-tests)
  return()
endif()

if(NOT TARGET OBS::libobs)
  find_package(libobs CONFIG REQUIRED)
endif()
if(NOT TARGET OBS::obs-frontend-api)
  find_package(obs-frontend-api CONFIG REQUIRED)
endif()
find_package(Qt6 REQUIRED COMPONENTS Widgets Core)

add_library(task-dock MODULE src/plugin.cpp src/logic.hpp)
target_link_libraries(task-dock PRIVATE OBS::libobs OBS::obs-frontend-api Qt6::Widgets Qt6::Core)
target_compile_features(task-dock PRIVATE cxx_std_17)
set_target_properties(task-dock PROPERTIES PREFIX "" CXX_VISIBILITY_PRESET hidden)

if(MSVC)
  target_compile_options(task-dock PRIVATE /W4 /utf-8 /permissive-)
else()
  target_compile_options(task-dock PRIVATE -Wall -Wextra -Wpedantic)
endif()

if(WIN32)
  install(TARGETS task-dock RUNTIME DESTINATION obs-plugins/64bit LIBRARY DESTINATION obs-plugins/64bit)
elseif(APPLE)
  message(FATAL_ERROR "macOS packaging/signing is not provided in this release.")
else()
  install(TARGETS task-dock LIBRARY DESTINATION lib/obs-plugins)
endif()
